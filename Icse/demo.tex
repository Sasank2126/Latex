\section{Matrix}

\begin{enumerate}
		\item If $A = \begin{pmatrix} 5 & a \\ b & 0 \end{pmatrix}$ is a symmetric matrix, show that $a = b$.
        \item Without expanding at any stage, find the value of:    \\

        $ \myvec{ a & b & c \\ a + 2x & b + 2y & c + 2z \\ x & y & z }. $
\item
    Use properties of determinants to solve for $x$:
 
		$ \myvec{ x + a & b & c \\ c & x + b & a \\ a & b & x + c } = 0, \quad \text{and} \, x \neq 0. $


		\item Using elementary transformations, find the inverse of the matrix:
        $ \myvec{ 0 & 1 & 2 \\ 1 & 2 & 3 \\ 3 & 1 & 1 }. $

	\item Using elementary transformations, find the inverse of the matrix:
        $ \myvec{ 0 & 1 & 2 \\ 1 & 2 & 3 \\ 3 & 1 & 1 }. $

	\end{enumerate}


\section{Triogonometery}

\begin{enumerate}

		\item Solve: $3 \tan^{-1}x + \cot^{-1}x = \pi$.
		\item If $\tan^{-1}a + \tan^{-1}b + \tan^{-1}c = \pi$, prove that $a + b + c = abc$.
	\item Solve: $\sin(x - y) = \sin x \cdot \tan^{-1}\left(\frac{dy}{dx}\right)$.
\end{enumerate}


\section{Differentiation}

\begin{enumerate}

		\item Find the value of constant $k$ such that the function $f(x)$ defined as:
        $ f(x) = \begin{cases} \frac{x^2 - 1}{x + 1}, & x \neq -1 \\ k, & x = -1 \end{cases} $
        is continuous at $x = -1$.
		\item Find the differential equation of the family of concentric circles $x^2 + y^2 = a^2$.
        \item Show that the function
        $ f(x) = \begin{cases} x^2, & x \leq 1 \\ 2x + 1, & x > 1 \end{cases} $
        is continuous at $x = 1$ but not differentiable.
        \item Verify Rolle's Theorem for the function $f(x) = e^{\sin x}$ on the interval $[0, \pi]$.
   
    \item
    If $x = \tan(\log y)$, prove that $(1 + x^2) \frac{d^2y}{dx^2} + (2x - a) \frac{dy}{dx} = 0$.
\item The population of a town grows at the rate of 10\% per year. Using a differential equation, find how long it will take for the population to grow four times.

\end{enumerate}

\section{Geometry}

	\item Find the approximate change in the volume $V$ of a cube of side $x$ meters caused by decreasing the side by 1\%.

        \item Water is dripping out from a conical funnel of semi-vertical angle $\theta$ at the uniform rate of 2 cm$^2$/sec through a tiny hole at the vertex. When the slant height of the water level is 4 cm, find the rate of decrease of the slant height of the water.

	\item

    \begin{enumerate}
        \item The probability distribution of $X$.
        \item The mean of $X$.
        \item The variance of $X$.
    \end{enumerate}

    \item Draw a rough sketch of the curve and find the area of the region bounded by the curve $y = 8x$ and the line $x = 2$.
        \item Sketch the graph of $y = |x + 4|$. Using integration, find the area of the region bounded by the curve $y = |x + 4|$ and $x = -6$ and $x = 0$.
    \item Find the coefficient of correlation from the regression lines:
        $ x - 2y + 3 = 0 \quad \text{and} \quad 4x - 5y + 1 = 0. $
		\end{enumerate}

\section{Probability}

\begin{enumerate}



\item If $A$ and $B$ are events such that $P(A) = \frac{1}{4}$, $P(B) = \frac{1}{2}$, and $P(A \cap B) = \frac{1}{8}$, then find:
        \begin{enumerate}
            \item $P(A|B)$
            \item $P(B|A)$
        \end{enumerate}
        \item In a race, the probabilities of $A$ and $B$ winning the race are $\frac{1}{3}$ and $\frac{1}{5}$ respectively. Find the probability of neither of them winning the race.
  

    \item    A speaks truth in 60\% of cases, while B speaks truth in 40\% of cases. In what percentage of cases are they likely to contradict each other in stating the same fact?

	    \item Find the line of regression of $y$ on $x$ from the following table:

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
        \hline
        x & 1 & 2 & 3 & 4 & 5 \\
        \hline
        y & 7 & 6 & 5 & 4 & 3 \\
        \hline
        \end{tabular} 
        \end{center}


        Hence, estimate the value of $y$ when $x = 6$.
        \item From the given data: 



\begin{center}
\begin{tabular}{ |c|c|c| } 


		
        \hline
        \text{Variable} & \text{Mean} & \text{Standard Deviation} \\
        \hline
        x & 6 & 4 \\
        y & 8 & 6 \\
        \hline
        \end{tabular} 
        \end{center}
        and correlation coefficient $r = 0.5$. Find:
\begin{enumerate}        
            \item Regression coefficients $b_{xy}$ and $b_{yx}$.
            \item Regression line $x$ on $y$.
            \item Most likely value of $x$ when $y = 14$.
\end{enumerate}

\end{enumerate}

\section{Vectors}

\begin{enumerate}
		

\item Find $\lambda$ if the scalar projection of $\vec{a} = \lambda \hat{i} + \hat{j} + 4\hat{k}$ on $\vec{b} = 2\hat{i} + 6\hat{j} + 3\hat{k}$ is 4 units.
        \item The Cartesian equation of a line is: $2x - 3 = 3y + 1 = 5 - 6z$. Find the vector equation of a line passing through $(7, -5, 0)$ and parallel to the given line.
        \item Find the equation of the plane through the intersection of the planes $\vec{r} \cdot (\hat{i} + 3\hat{j} - \hat{k}) = 9$ and $\vec{r} \cdot (2\hat{i} - \hat{j} + \hat{k}) = 3$ and passing through the origin.
    
        \item If $A$, $B$, and $C$ are three non-collinear points with position vectors $\vec{a}$, $\vec{b}$, and$\vec{c}$ respectively, show that the length of the perpendicular from $C$ on $AB$ is given by:
        $ \frac{|\vec{a} \times \vec{b}|}{|\vec{a} - \vec{b}|}. $
        \item Show that the four points $A$, $B$, $C$, and $D$ with position vectors $4\hat{i} + 5\hat{j} + \hat{k}$, $-\hat{j} - \hat{k}$, $3\hat{i} + 9\hat{j} + 4\hat{k}$, and $4(-\hat{i} + \hat{j} + \hat{k})$ respectively, are coplanar.
    

	\item Find the image of a point having position vector $3\hat{i} - 2\hat{j} + \hat{k}$ in the plane $\vec{r} \cdot (3\hat{i} - \hat{j} + 4\hat{k}) = 2$.

\end{enumerate}




\section{Calculus}


\begin{enumerate}


		\item Given the total cost function for $x$ units of a commodity as:
        $ C(x) = x^3 + 3x^2 - 16x + 2. $
        Find:
        \begin{enumerate}
            \item Marginal cost function.
            \item Average cost function.
        \end{enumerate}
        

		\item The average cost function associated with producing and marketing $x$ units of an item is given by:
        $ AC = 2x - 11 + \frac{1}{x}. $
        Find the range of values of the output $x$, for which $AC$ is increasing.

	\item A manufacturer’s marginal cost function is $C'(x) = \sqrt{x} + 2$. Find the cost involved to increase production from 100 units to 300 units.
\end{enumerate}

\section{Linear Programming}

\begin{enumerate}

		\item A product can be manufactured at a total cost $C(x) = x^2 + 100x + 40$, where $x$ is the number of units produced. The price at which each unit can be sold is given by $P = 200 - \frac{x}{2}$. Determine the production level $x$ at which the profit is maximum. What is the price per unit and total profit at this level of production?
        
			\item
    A manufacturing company makes two types of teaching aids $A$ and $B$ of Mathematics for Class X. Each type of $A$ requires 9 labor hours for fabricating and 1 labor hour for finishing. Each type of $B$ requires 12 labor hours for fabricating and 3 labor hours for finishing. For fabricating and finishing, the maximum labor hours available per week are 180 and 30 respectively. The company makes a profit of 80 Rupees on each piece of type $A$ and 120 Rupees on each piece of type $B$. How many pieces of type $A$ and type $B$ should be manufactured per week to get a maximum profit? Formulate this as a Linear Programming Problem and solve it. Identify the feasible region from the rough sketch.
\end{enumerate}
